### Script de teste automatizado

```typescript
// test/e2e/multi-user-isolation.test.ts
describe('Multi-user isolation', () => {
  let chrisClient: SupabaseClient;
  let guilgerClient: SupabaseClient;

  beforeAll(async () => {
    chrisClient = createAuthenticatedClient('chris_rodrigues');
    guilgerClient = createAuthenticatedClient('guilger_oliveira');
  });

  test('guilger cannot see chris memories', async () => {
    // Chris insere uma memoria
    await chrisClient
      .from('user_memories')
      .insert({
        user_id: CHRIS_UUID,
        key: 'test_isolation',
        value: 'chris_secret',
      });

    // Guilger tenta ler
    const { data } = await guilgerClient
      .from('user_memories')
      .select('*')
      .eq('key', 'test_isolation');

    expect(data).toEqual([]);
  });

  test('guilger cannot see chris debate sessions', async () => {
    const { data } = await guilgerClient
      .from('debate_sessions')
      .select('*');

    // Nenhuma sessao de chris visivel
    const chrisSessions = data?.filter(
      (s) => s.user_id === CHRIS_UUID
    );
    expect(chrisSessions).toEqual([]);
  });

  test('guilger can see shared minds', async () => {
    const { data } = await guilgerClient
      .from('minds_metadata')
      .select('*');

    expect(data?.length).toBeGreaterThanOrEqual(27);
  });

  test('chris clone unchanged after guilger onboarding',
    async () => {
      const beforeOnboarding = await getCloneFidelity(
        CHRIS_UUID
      );

      // Simular onboarding de guilger
      await performOnboarding(guilgerClient, GUILGER_UUID);

      const afterOnboarding = await getCloneFidelity(
        CHRIS_UUID
      );

      expect(afterOnboarding).toEqual(beforeOnboarding);
    }
  );
});
```
