## Story
**As a** sistema teamAI,
**I want** ter o debate_engine.py usando a Anthropic API (Claude) em vez de Gemini,
**so that** os debates entre minds usem o modelo com maior context window e melhor qualidade para raciocinio complexo.

## Acceptance Criteria
1. `debate_engine.py` usa o SDK `anthropic` para chamadas de API
2. Suporta streaming de respostas via `client.messages.stream()`
3. Aceita API key do usuario como parametro (nao hardcoded)
4. Suporta selecao de modelo (default: claude-sonnet-4-20250514)
5. Gera resposta de um mind dado system_prompt + KB + mensagem do usuario
6. Trata erros de API (rate limit, token overflow, auth failure) com mensagens claras
7. Conta tokens de input/output por turno
8. Testes unitarios cobrindo happy path e erros

### Assinatura da classe DebateEngine

```python
from anthropic import Anthropic
from collections.abc import AsyncIterator
from dataclasses import dataclass


@dataclass
class TurnResult:
    """Resultado de um turno de debate."""
    mind_slug: str
    content: str
    input_tokens: int
    output_tokens: int
    model: str


class DebateEngine:
    """Motor de debate usando Anthropic API."""

    DEFAULT_MODEL = "claude-sonnet-4-20250514"
    MAX_TOKENS = 4096

    def __init__(
        self,
        api_key: str,
        model: str | None = None,
    ) -> None:
        self.client = Anthropic(api_key=api_key)
        self.model = model or self.DEFAULT_MODEL

    def generate_response(
        self,
        mind_slug: str,
        system_prompt: str,
        kb_context: str,
        user_message: str,
        max_tokens: int | None = None,
    ) -> TurnResult:
        """Gera resposta de um mind para uma mensagem."""
        full_system = f"{system_prompt}\n\n## Knowledge Base\n{kb_context}"
        response = self.client.messages.create(
            model=self.model,
            max_tokens=max_tokens or self.MAX_TOKENS,
            system=full_system,
            messages=[{"role": "user", "content": user_message}],
        )
        return TurnResult(
            mind_slug=mind_slug,
            content=response.content[0].text,
            input_tokens=response.usage.input_tokens,
            output_tokens=response.usage.output_tokens,
            model=self.model,
        )

    def generate_response_stream(
        self,
        mind_slug: str,
        system_prompt: str,
        kb_context: str,
        user_message: str,
        max_tokens: int | None = None,
    ) -> AsyncIterator[str]:
        """Gera resposta com streaming."""
        full_system = f"{system_prompt}\n\n## Knowledge Base\n{kb_context}"
        with self.client.messages.stream(
            model=self.model,
            max_tokens=max_tokens or self.MAX_TOKENS,
            system=full_system,
            messages=[{"role": "user", "content": user_message}],
        ) as stream:
            for text in stream.text_stream:
                yield text
```

### Assinatura do MindLoader

```python
from pathlib import Path


class MindLoader:
    """Carrega system_prompt e KB de um mind."""

    def __init__(self, minds_base_path: str | Path) -> None:
        self.base = Path(minds_base_path)

    def load_system_prompt(self, mind_slug: str) -> str:
        """Carrega o system prompt principal do mind."""
        prompt_dir = self.base / mind_slug / "system_prompts"
        prompt_file = prompt_dir / "main.md"
        if not prompt_file.exists():
            msg = f"System prompt not found: {prompt_file}"
            raise FileNotFoundError(msg)
        return prompt_file.read_text()

    def load_kb(self, mind_slug: str) -> str:
        """Carrega todos os chunks de KB do mind."""
        kb_dir = self.base / mind_slug / "kb"
        if not kb_dir.exists():
            return ""
        chunks = sorted(kb_dir.glob("*.md"))
        return "\n\n---\n\n".join(f.read_text() for f in chunks)
```

### Error handling

```python
from anthropic import (
    APIError,
    AuthenticationError,
    BadRequestError,
    RateLimitError,
)


class DebateEngineError(Exception):
    """Erro base do debate engine."""


class TokenOverflowError(DebateEngineError):
    """Context window excedido."""


class RateLimitExceededError(DebateEngineError):
    """Rate limit da API atingido."""


class InvalidAPIKeyError(DebateEngineError):
    """API key invalida ou expirada."""


# No generate_response():
try:
    response = self.client.messages.create(...)
except RateLimitError as e:
    raise RateLimitExceededError(
        "Rate limit atingido. Aguarde alguns segundos e tente novamente."
    ) from e
except BadRequestError as e:
    if "token" in str(e).lower():
        raise TokenOverflowError(
            f"Context window excedido para mind '{mind_slug}'. "
            "Considere usar KB compression."
        ) from e
    raise DebateEngineError(f"Erro na requisicao: {e}") from e
except AuthenticationError as e:
    raise InvalidAPIKeyError(
        "API key invalida ou expirada. Verifique sua key Anthropic."
    ) from e
```

### Gotchas
- O SDK anthropic usa `system` como parametro separado em `messages.create()`, nao como mensagem no array
- Streaming usa `client.messages.stream()` (context manager), nao `create(stream=True)`
- Token counting vem no `response.usage` — nao precisa de tokenizer local
- A API key NUNCA deve ser hardcoded — sempre recebida como parametro ou env var
- O modelo default deve ser claude-sonnet-4 (melhor custo-beneficio para debates)
