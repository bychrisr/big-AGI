## Story
**As a** usuario do teamAI,
**I want** que o system prompt de cada mind seja carregado apenas uma vez por sessao de debate,
**so that** turnos subsequentes consumam muito menos tokens enviando apenas o delta da conversa.

## Acceptance Criteria
1. Classe DebateSession gerencia estado de uma sessao de debate
2. System prompt + KB carregados apenas no primeiro turno de cada mind (caching)
3. Turnos seguintes enviam apenas historico resumido + nova mensagem (delta)
4. Sessao suporta multiplos minds debatendo em sequencia
5. Sessao rastreia total de tokens consumidos (input + output) por mind e total
6. Sessao pode ser serializada para dict (para persistencia futura no Supabase)
7. Testes cobrindo: primeiro turno vs turnos seguintes; multiplos minds; serializacao

### Implementacao da DebateSession

```python
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timezone

from debate_engine.engine import DebateEngine, TurnResult
from debate_engine.mind_loader import MindLoader


@dataclass
class Turn:
    """Registro de um turno no debate."""
    mind_slug: str
    role: str  # "user" | "assistant"
    content: str
    input_tokens: int
    output_tokens: int
    timestamp: str


@dataclass
class MindTokenUsage:
    """Uso de tokens acumulado por mind."""
    input_tokens: int = 0
    output_tokens: int = 0

    @property
    def total(self) -> int:
        return self.input_tokens + self.output_tokens


class DebateSession:
    """Gerencia uma sessao de debate com session caching."""

    def __init__(
        self,
        topic: str,
        minds: list[str],
        user_id: str,
        engine: DebateEngine,
        mind_loader: MindLoader,
        session_id: str | None = None,
    ) -> None:
        self.session_id = session_id or str(uuid.uuid4())
        self.topic = topic
        self.minds = minds
        self.user_id = user_id
        self.engine = engine
        self.mind_loader = mind_loader

        self.initialized: dict[str, bool] = {}
        self.history: list[Turn] = []
        self.token_usage: dict[str, MindTokenUsage] = {
            m: MindTokenUsage() for m in minds
        }
        self.created_at = datetime.now(tz=timezone.utc).isoformat()

    def run_turn(
        self,
        mind_slug: str,
        user_message: str,
    ) -> TurnResult:
        """Executa um turno para um mind especifico."""
        if mind_slug not in self.minds:
            msg = f"Mind '{mind_slug}' nao faz parte desta sessao"
            raise ValueError(msg)

        system_prompt = self.mind_loader.load_system_prompt(mind_slug)
        kb_context = self.mind_loader.load_kb(
            mind_slug, topic=self.topic
        )
        prompt = self._get_prompt_for_turn(
            mind_slug, user_message, system_prompt, kb_context
        )

        result = self.engine.generate_response(
            mind_slug=mind_slug,
            system_prompt=system_prompt if not self.initialized.get(mind_slug) else "",
            kb_context=kb_context if not self.initialized.get(mind_slug) else "",
            user_message=prompt,
        )

        self.initialized[mind_slug] = True
        self._record_turn(mind_slug, "user", user_message, 0, 0)
        self._record_turn(
            mind_slug, "assistant", result.content,
            result.input_tokens, result.output_tokens,
        )
        self._update_token_usage(mind_slug, result)

        return result

    def _get_prompt_for_turn(
        self,
        mind_slug: str,
        user_message: str,
        system_prompt: str,
        kb_context: str,
    ) -> str:
        """Retorna prompt otimizado: completo no 1o turno, delta nos seguintes."""
        if not self.initialized.get(mind_slug):
            return user_message

        # Delta: apenas historico recente + nova mensagem
        recent = self._get_recent_history(mind_slug, max_turns=4)
        if recent:
            context = "\n".join(
                f"[{t.role}]: {t.content[:500]}" for t in recent
            )
            return f"Contexto recente:\n{context}\n\nNova mensagem: {user_message}"
        return user_message

    def _get_recent_history(
        self,
        mind_slug: str,
        max_turns: int = 4,
    ) -> list[Turn]:
        """Retorna turnos recentes de um mind."""
        mind_turns = [t for t in self.history if t.mind_slug == mind_slug]
        return mind_turns[-max_turns:]

    def _record_turn(
        self,
        mind_slug: str,
        role: str,
        content: str,
        input_tokens: int,
        output_tokens: int,
    ) -> None:
        self.history.append(Turn(
            mind_slug=mind_slug,
            role=role,
            content=content,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            timestamp=datetime.now(tz=timezone.utc).isoformat(),
        ))

    def _update_token_usage(
        self,
        mind_slug: str,
        result: TurnResult,
    ) -> None:
        usage = self.token_usage[mind_slug]
        usage.input_tokens += result.input_tokens
        usage.output_tokens += result.output_tokens

    @property
    def total_tokens(self) -> int:
        return sum(u.total for u in self.token_usage.values())

    def to_dict(self) -> dict:
        """Serializa sessao para persistencia."""
        return {
            "session_id": self.session_id,
            "topic": self.topic,
            "minds": self.minds,
            "user_id": self.user_id,
            "initialized": self.initialized,
            "history": [
                {
                    "mind_slug": t.mind_slug,
                    "role": t.role,
                    "content": t.content,
                    "input_tokens": t.input_tokens,
                    "output_tokens": t.output_tokens,
                    "timestamp": t.timestamp,
                }
                for t in self.history
            ],
            "token_usage": {
                k: {"input_tokens": v.input_tokens, "output_tokens": v.output_tokens}
                for k, v in self.token_usage.items()
            },
            "created_at": self.created_at,
            "total_tokens": self.total_tokens,
        }

    @classmethod
    def from_dict(
        cls,
        data: dict,
        engine: DebateEngine,
        mind_loader: MindLoader,
    ) -> "DebateSession":
        """Reconstroi sessao a partir de dict."""
        session = cls(
            topic=data["topic"],
            minds=data["minds"],
            user_id=data["user_id"],
            engine=engine,
            mind_loader=mind_loader,
            session_id=data["session_id"],
        )
        session.initialized = data["initialized"]
        session.history = [
            Turn(**t) for t in data["history"]
        ]
        session.token_usage = {
            k: MindTokenUsage(**v) for k, v in data["token_usage"].items()
        }
        session.created_at = data["created_at"]
        return session
```

### Economia de tokens esperada

| Cenario | Sem caching | Com caching | Economia |
|---------|-------------|-------------|----------|
| 1 mind, 5 turnos | 5x (SP+KB+msg) | 1x (SP+KB+msg) + 4x (delta) | ~70% |
| 3 minds, 3 turnos cada | 9x (SP+KB+msg) | 3x (SP+KB+msg) + 6x (delta) | ~60% |
| Growth Council (5 minds, 2 turnos) | 10x completo | 5x completo + 5x delta | ~45% |

### Gotchas
- O caching e por SESSAO, nao global. Cada DebateSession nova carrega system prompts do zero.
- A Anthropic API nao tem cache de system prompt nativo (diferente de prompt caching beta). Aqui, "caching" significa nao RE-ENVIAR o system prompt nos turnos seguintes.
- O historico recente (_get_recent_history) e truncado a 500 chars por turno para controlar tokens.
- `from_dict()` precisa receber engine e mind_loader porque nao sao serializaveis.
- A serializacao nao inclui o DebateEngine nem MindLoader — eles sao reinjetados no from_dict().
