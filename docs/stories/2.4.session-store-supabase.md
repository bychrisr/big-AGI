### Implementacao do SessionStore

```python
from supabase import create_client, Client


class SessionStore:
    """Persistencia de sessoes de debate no Supabase."""

    TABLE = "debate_sessions"

    def __init__(self, supabase_url: str, supabase_key: str) -> None:
        self.client: Client = create_client(supabase_url, supabase_key)

    def save(self, session_data: dict, user_id: str) -> None:
        """Salva ou atualiza uma sessao de debate."""
        row = {
            "user_id": user_id,
            "session_id": session_data["session_id"],
            "topic": session_data["topic"],
            "minds": session_data["minds"],
            "turns": session_data["history"],
            "token_usage": session_data["token_usage"],
            "total_tokens": session_data["total_tokens"],
            "initialized": session_data["initialized"],
        }
        self.client.table(self.TABLE).upsert(
            row, on_conflict="session_id"
        ).execute()

    def load(self, session_id: str) -> dict | None:
        """Carrega uma sessao pelo session_id."""
        result = (
            self.client.table(self.TABLE)
            .select("*")
            .eq("session_id", session_id)
            .single()
            .execute()
        )
        if not result.data:
            return None
        return self._row_to_session_data(result.data)

    def list_sessions(
        self,
        user_id: str,
        status: str = "active",
        limit: int = 20,
        offset: int = 0,
    ) -> list[dict]:
        """Lista sessoes do usuario com paginacao."""
        result = (
            self.client.table(self.TABLE)
            .select("session_id, topic, minds, total_tokens, status, created_at, updated_at")
            .eq("user_id", user_id)
            .eq("status", status)
            .order("created_at", desc=True)
            .range(offset, offset + limit - 1)
            .execute()
        )
        return result.data or []

    def delete(self, session_id: str) -> bool:
        """Remove uma sessao."""
        result = (
            self.client.table(self.TABLE)
            .delete()
            .eq("session_id", session_id)
            .execute()
        )
        return bool(result.data)

    def archive(self, session_id: str) -> None:
        """Arquiva uma sessao (soft delete)."""
        self.client.table(self.TABLE).update(
            {"status": "archived"}
        ).eq("session_id", session_id).execute()

    @staticmethod
    def _row_to_session_data(row: dict) -> dict:
        """Converte row do Supabase para formato DebateSession."""
        return {
            "session_id": row["session_id"],
            "topic": row["topic"],
            "minds": row["minds"],
            "user_id": str(row["user_id"]),
            "history": row["turns"],
            "token_usage": row["token_usage"],
            "initialized": row["initialized"],
            "created_at": row["created_at"],
        }
```

```python
    result = self.engine.generate_response(...)
    # ... record turn, update tokens ...

    # Auto-save se store configurado

        self.session_store.save(self.to_dict(), self.user_id)

    return result

# Classmethod para resume:
@classmethod
def resume(
    cls,
    session_id: str,
    engine: DebateEngine,
    mind_loader: MindLoader,
    session_store: SessionStore,
) -> "DebateSession":
    """Retoma uma sessao salva no Supabase."""
    data = session_store.load(session_id)
    if data is None:
        msg = f"Sessao nao encontrada: {session_id}"
        raise ValueError(msg)
    session = cls.from_dict(data, engine, mind_loader)
    session.session_store = session_store
    return session
```

### Gotchas
- O campo `turns` e JSONB no Supabase — armazena o array de dicts do historico. Cuidado com o tamanho para sessoes longas.
- RLS depende de `auth.uid()` — o client Supabase deve ser inicializado com o JWT do usuario, nao com a service_role key.
- `upsert` com `on_conflict="session_id"` permite save idempotente — chamado multiplas vezes nao duplica.
- O campo `minds` e `TEXT[]` (array do Postgres), nao JSONB. O supabase-py converte automaticamente lista Python <-> Postgres array.
- Supabase free tier tem limite de 500MB — monitorar tamanho da tabela com sessoes longas.
- O `delete()` e hard delete. Para soft delete, usar `archive()` que muda status para 'archived'.
