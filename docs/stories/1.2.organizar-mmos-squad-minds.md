### Script create-minds.sh

```bash
#!/bin/bash
set -euo pipefail

MINDS_DIR="squads-base/mmos-squad/minds"

MINDS=(
  alan_nicolas
  alex_hormozi
  dickie_bush
  elon_musk
  eugene_schwartz
  jeff_patton
  justin_welsh
  marty_cagan
  naval_ravikant
  paul_graham
  sahil_lavingia
  sam_altman
  seth_godin
  simon_sinek
)

for mind in "${MINDS[@]}"; do
  mkdir -p "$MINDS_DIR/$mind"/{kb,system_prompts}
  touch "$MINDS_DIR/$mind/kb/.gitkeep"
  touch "$MINDS_DIR/$mind/system_prompts/.gitkeep"
  echo "Created: $mind"
done

echo "Total minds created: ${#MINDS[@]}"
```

### Script validate-minds.sh

```bash
#!/bin/bash
set -euo pipefail

MINDS_DIR="squads-base/mmos-squad/minds"
errors=0

for mind_dir in "$MINDS_DIR"/*/; do
  mind=$(basename "$mind_dir")
  if [ ! -f "$mind_dir/metadata.yaml" ]; then
    echo "ERROR: $mind is missing metadata.yaml"
    errors=$((errors + 1))
  fi
  if [ ! -d "$mind_dir/kb" ]; then
    echo "ERROR: $mind is missing kb/ directory"
    errors=$((errors + 1))
  fi
  if [ ! -d "$mind_dir/system_prompts" ]; then
    echo "ERROR: $mind is missing system_prompts/ directory"
    errors=$((errors + 1))
  fi
done

if [ $errors -eq 0 ]; then
  echo "All minds validated successfully"
else
  echo "Found $errors errors"
  exit 1
fi
```
