## Story
**As a** frontend do teamAI (big-AGI fork),
**I want** ter uma route /api/debate que autentica via Supabase JWT e orquestra o debate_engine Python,
**so that** o browser possa iniciar e gerenciar debates entre minds com streaming de respostas.

## Acceptance Criteria
1. Route Handler `POST /api/debate` criada no big-AGI fork (App Router)
2. Autenticacao via Supabase JWT no header Authorization
3. Request body aceita: topic, minds[], user_message, session_id (opcional)
4. Route chama debate_engine Python como subprocess
5. Resposta retornada como streaming (ReadableStream) para o frontend
6. Erros de autenticacao retornam 401; erros de validacao retornam 400; erros do engine retornam 500
7. Testes cobrindo: autenticacao, validacao, happy path
8. Tipos TypeScript completos para request/response

// Request body
interface DebateRequest {
  topic: string;
  minds: string[];
  user_message: string;
  session_id?: string;
  model?: string;
  max_tokens?: number;
}

// Response (streaming JSON lines)
interface DebateStreamChunk {
  type: 'text' | 'token_usage' | 'error' | 'done';
  mind_slug: string;
  content?: string;
  input_tokens?: number;
  output_tokens?: number;
  error?: string;
}

// Error response
interface DebateErrorResponse {
  error: string;
  code: 'AUTH_REQUIRED' | 'AUTH_INVALID' | 'VALIDATION_ERROR' | 'ENGINE_ERROR';
}

### Implementacao da route

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';
import { spawn } from 'child_process';

const SUPABASE_URL = process.env.SUPABASE_URL!;
const SUPABASE_ANON_KEY = process.env.SUPABASE_ANON_KEY!;
const DEBATE_ENGINE_PATH = process.env.DEBATE_ENGINE_PATH
  || '../teamai/squads-base/mmos-squad';

export async function POST(req: NextRequest): Promise<Response> {
  // 1. Autenticacao
  const authHeader = req.headers.get('Authorization');
  if (!authHeader?.startsWith('Bearer ')) {
    return NextResponse.json(
      { error: 'Authorization header required', code: 'AUTH_REQUIRED' },
      { status: 401 },
    );
  }

  const token = authHeader.slice(7);
  const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    global: { headers: { Authorization: `Bearer ${token}` } },
  });

  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) {
    return NextResponse.json(
      { error: 'Invalid or expired token', code: 'AUTH_INVALID' },
      { status: 401 },
    );
  }

  // 2. Validacao do body
  const body: DebateRequest = await req.json();
  if (!body.topic || !body.minds?.length || !body.user_message) {
    return NextResponse.json(
      {
        error: 'Missing required fields: topic, minds, user_message',
        code: 'VALIDATION_ERROR',
      },
      { status: 400 },
    );
  }

  // 3. Chamar debate_engine como subprocess
  try {
    const stream = await runDebateEngine({
      ...body,
      user_id: user.id,
      api_key: await getUserApiKey(supabase, user.id),
    });
    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        'Connection': 'keep-alive',
      },
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : 'Unknown error';
    return NextResponse.json(
      { error: message, code: 'ENGINE_ERROR' },
      { status: 500 },
    );
  }
}

async function getUserApiKey(
  supabase: ReturnType<typeof createClient>,
  userId: string,
): Promise<string> {
  const { data, error } = await supabase
    .from('user_api_keys')
    .select('encrypted_key')
    .eq('user_id', userId)
    .eq('provider', 'anthropic')
    .single();

  if (error || !data) {
    throw new Error('Anthropic API key not configured for this user');
  }
  return data.encrypted_key;
}

interface EngineInput extends DebateRequest {
  user_id: string;
  api_key: string;
}

function runDebateEngine(input: EngineInput): ReadableStream {
  const encoder = new TextEncoder();

  return new ReadableStream({
    start(controller) {
      const proc = spawn('python', [
        '-m', 'debate_engine.cli',
        '--topic', input.topic,
        '--minds', input.minds.join(','),
        '--message', input.user_message,
        '--user-id', input.user_id,
        ...(input.session_id ? ['--session-id', input.session_id] : []),
        ...(input.model ? ['--model', input.model] : []),
      ], {
        cwd: DEBATE_ENGINE_PATH,
        env: {
          ...process.env,
          ANTHROPIC_API_KEY: input.api_key,
        },
      });

      proc.stdout.on('data', (data: Buffer) => {
        controller.enqueue(encoder.encode(`data: ${data.toString()}\n\n`));
      });

      proc.stderr.on('data', (data: Buffer) => {
        const errorChunk: DebateStreamChunk = {
          type: 'error',
          mind_slug: '',
          error: data.toString(),
        };
        controller.enqueue(
          encoder.encode(`data: ${JSON.stringify(errorChunk)}\n\n`)
        );
      });

      proc.on('close', (code) => {
        const doneChunk: DebateStreamChunk = {
          type: 'done',
          mind_slug: '',
          content: code === 0 ? 'completed' : `exited with code ${code}`,
        };
        controller.enqueue(
          encoder.encode(`data: ${JSON.stringify(doneChunk)}\n\n`)
        );
        controller.close();
      });
    },
  });
}
```

### CLI do debate_engine (Python side)

Para a route funcionar, o debate_engine precisa de um CLI que aceite argumentos e imprima JSON lines no stdout:

```python
# squads-base/mmos-squad/debate_engine/cli.py
import argparse
import json
import os
import sys

from debate_engine.engine import DebateEngine
from debate_engine.mind_loader import MindLoader
from debate_engine.session import DebateSession


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--topic", required=True)
    parser.add_argument("--minds", required=True)
    parser.add_argument("--message", required=True)
    parser.add_argument("--user-id", required=True)
    parser.add_argument("--session-id", default=None)
    parser.add_argument("--model", default=None)
    args = parser.parse_args()

    api_key = os.environ["ANTHROPIC_API_KEY"]
    minds = args.minds.split(",")

    engine = DebateEngine(api_key=api_key, model=args.model)
    loader = MindLoader(minds_base_path="minds")

    session = DebateSession(
        topic=args.topic,
        minds=minds,
        user_id=args.user_id,
        engine=engine,
        mind_loader=loader,
        session_id=args.session_id,
    )

    for mind in minds:
        result = session.run_turn(mind, args.message)
        chunk = {
            "type": "text",
            "mind_slug": result.mind_slug,
            "content": result.content,
            "input_tokens": result.input_tokens,
            "output_tokens": result.output_tokens,
        }
        print(json.dumps(chunk), flush=True)

    done = {"type": "done", "mind_slug": "", "content": "completed"}
    print(json.dumps(done), flush=True)


if __name__ == "__main__":
    main()
```

### Gotchas
- A route usa `spawn` (child_process) para chamar Python — isso EXIGE Node.js runtime, nao Edge Runtime. Nao adicionar `export const runtime = 'edge'`.
- A API key do usuario vem do Supabase (tabela user_api_keys), NAO do localStorage. Isso e diferente do fluxo nativo do big-AGI.
- O streaming usa Server-Sent Events (SSE) com format `data: ...\n\n`. O frontend precisa de um EventSource ou fetch com ReadableStream.
- O DEBATE_ENGINE_PATH deve apontar para o diretorio do mmos-squad no repo teamai, que fica FORA do diretorio do big-AGI fork. Em Docker, ambos estarao montados.
- Seguranca: NUNCA passar argumentos do usuario diretamente no shell. Usar array de args no spawn() (ja feito acima).
- O subprocess herda PATH do processo Node — Python deve estar acessivel. Em Docker, instalar Python no mesmo container ou usar container separado.
